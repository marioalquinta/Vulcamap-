    <script type="module">
        import { initializeApp } from "https://gstatic.com";
        import { getFirestore, collection, addDoc, onSnapshot } from "https://gstatic.com";

        // CONFIGURACIÓN DE FIREBASE
        const firebaseConfig = {
            apiKey: "TU_API_KEY",
            authDomain: "TU_://firebaseapp.com",
            projectId: "TU_PROYECTO",
            storageBucket: "TU_://appspot.com",
            messagingSenderId: "TU_ID",
            appId: "TU_APP_ID"
        };

        const app = initializeApp(firebaseConfig);
        const db = getFirestore(app);
        const vulcasRef = collection(db, "vulcanizaciones");

        const map = L.map('map').setView([-33.4180, -70.6650], 14);
        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { attribution: '© OpenStreetMap' }).addTo(map);

        let tempCoords = null;
        let userLocationMarker = null;
        let selectedRating = 5;
        let localVulcasRawData = [];
        let filters = { text: '', only24h: false, highStars: false };

        // Lógica de estrellas del formulario
        const stars = document.querySelectorAll('.star');
        stars.forEach(star => {
            star.addEventListener('click', (e) => {
                selectedRating = parseInt(e.target.getAttribute('data-value'));
                updateStarUI(selectedRating);
            });
        });

        function updateStarUI(rating) {
            stars.forEach(star => {
                const value = parseInt(star.getAttribute('data-value'));
                if (value <= rating) star.classList.add('active');
                else star.classList.remove('active');
            });
        }
        updateStarUI(5);

        // Escuchar base de datos en tiempo real
        onSnapshot(vulcasRef, (snapshot) => {
            localVulcasRawData = [];
            snapshot.forEach((doc) => {
                localVulcasRawData.push(doc.data());
            });
            window.applyFilters();
        });

        window.applyFilters = function() {
            map.eachLayer((layer) => {
                if (layer instanceof L.Marker && layer !== userLocationMarker) {
                    map.removeLayer(layer);
                }
            });

            const searchText = document.getElementById('search-input').value.toLowerCase();

            localVulcasRawData.forEach(data => {
                if (searchText && !data.name.toLowerCase().includes(searchText)) return;
                if (filters.only24h && data.schedule !== '24 Horas') return;
                if (filters.highStars && (data.rating || 5) < 4) return;

                createMarker(data);
            });
        }

        window.toggleFilter = function(type) {
            if (type === '24h') {
                filters.only24h = !filters.only24h;
                document.getElementById('btn-filter-24h').classList.toggle('active', filters.only24h);
            } else if (type === 'stars') {
                filters.highStars = !filters.highStars;
                document.getElementById('btn-filter-stars').classList.toggle('active', filters.highStars);
            }
            window.applyFilters();
        }
